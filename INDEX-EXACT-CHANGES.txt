2) Immediately before </head>, paste this ONE-TIME cache reset:

<script>
(() => {
  const KEY = 'ADGB_ASSET_RESET_V169';
  if (localStorage.getItem(KEY) === 'done') return;
  localStorage.setItem(KEY, 'done');

  try {
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.getRegistrations().then(registrations => {
        registrations.forEach(registration => {
          if (String(registration.scope || '').includes('/ADG-B-Report-Portal/')) {
            registration.unregister();
          }
        });
      });
    }
  } catch (e) {}

  try {
    if ('caches' in window) {
      caches.keys().then(keys => Promise.all(keys.map(key => caches.delete(key))));
    }
  } catch (e) {}
})();
</script>

3) AT THE VERY BOTTOM, REMOVE old raw-data.js / sticky-notes.js script lines
   and use ONLY these:

<script src="portal-v169.js"></script>
<script src="sticky-v169.js"></script>

Do not keep these old lines:
